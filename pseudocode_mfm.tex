\begin{algorithm}[H]
\caption{MFM: one life}
\begin{algorithmic}[1]
\Require task $\tau$; seeds $S$ (workspaces of earlier lives in the same world, possibly $\emptyset$)
\Require atom set $\mathcal{A}$ (the 28 primitive procedures); step budget $N$
\Ensure the workspace $W$, which is the record of this life and the seed of the next
\State $W \gets$ fresh directory \Comment{the model's cwd and its entire memory}
\State \Call{SeedPrevious}{$W, S$}
\State $W$/\file{AGENTS.md} $\gets$ \Call{Contract}{$\tau, \mathcal{A}$, previous $= [S \neq \emptyset]$}
\State write $W$/\file{skills/}, $W$/\file{references/}, $W$/\file{tools/episodes.py}
\State $T \gets$ new codex thread with cwd $W$ \Comment{one conversation; compacts at 100k tokens; cold start on overflow}
\State $Q \gets \langle\rangle$; $L \gets$ milestone ledger of $\tau$
\For{$s = 1 \dots N$}
    \State $o \gets$ env.observation()
    \State \Call{Observe}{$o, s$} \Comment{ground truth into memory; no model call}
    \State $a \gets$ \Call{NextAction}{$o, s$} \Comment{a model turn only when $Q$ is empty}
    \State env.step($a$)
\EndFor
\State close the open episode; write the report
\State \Return $W$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{The carry}
\begin{algorithmic}[1]
\Procedure{SeedPrevious}{$W, S$}
    \State $\mathit{lives} \gets$ every life reachable from $S$ (a seed may itself hold \file{previous/}), deduplicated, oldest first
    \For{$i, \mathit{src}$ \textbf{in} enumerate($\mathit{lives}$)}
        \State hard-link $\mathit{src}/*$ into $W$/\file{previous/life\_$i$/}, skipping harness-only files
        \State $\mathit{rows} \gets \mathit{rows} \cup \{ r \in \mathit{src}/\file{index.jsonl} : r.\mathit{life} \gets i,\ r.\mathit{paths} \gets \file{previous/life\_$i$/}\,r.\mathit{paths} \}$
    \EndFor
    \State $W$/\file{previous/index.jsonl} $\gets \mathit{rows}$
    \Statex \Comment{Nothing is digested for the model. The contract says \file{previous/} exists and that turn 1 is a workshop turn.}
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Observe: one environment step into the record}
\begin{algorithmic}[1]
\Procedure{Observe}{$o, s$}
    \State $F \gets L.\textsc{update}(o)$ \Comment{milestone rules over the stat stream}
    \If{$F \neq \emptyset$}
        \Log{\texttt{[EVENT] MILESTONE} $F$}; $\mathit{pending} \gets \mathit{pending} \cup F$
        \If{$Q \neq \langle\rangle$ and no macro or GUI at its head} $Q \gets \langle\rangle$ \Comment{the chain is ordered: re-plan} \EndIf
    \EndIf
    \State $E \gets L.\textsc{statEvents}(o)$; \textbf{log} $E$; $\mathit{pending} \gets \mathit{pending} \cup E$ \Comment{mine, craft, pick up}
    \State $p, h \gets$ position and health of $o$
    \If{$\lVert p - p_{\mathrm{prev}} \rVert_1 > 15$} \Comment{a teleport}
        \If{$h = h_{\mathrm{prev}}$ and coordinates scale by 8 and $p \neq$ spawn}
            \State $\mathit{dim} \gets$ Nether or Overworld; \Note{``a portal trip, not a death''}; $Q \gets \langle\rangle$
        \ElsIf{$h \ge 19$}
            \Note{``YOU DIED at $p_{\mathrm{prev}}$''}; $Q \gets \langle\rangle$
        \EndIf
    \ElsIf{$h_{\mathrm{prev}} - h \ge 3$ in one tick (or $\ge 4$ slowly)}
        \Note{``damage: health $h/20$ at $p$; find the danger before continuing''}; $Q \gets \langle\rangle$
        \Statex \Comment{harness reflexes are off in this run: the note stands, the runner does not act}
    \EndIf
    \Log{\texttt{[STATE]} $p$, pitch, yaw}; grid.observe($p$); $\mathit{pov} \gets o.\mathit{pov}$
    \State \Call{Guards}{$s, p$} \Comment{Nether watcher, lava, shaft, fall, water: each may note and clear $Q$}
    \State $\mathit{inv} \gets$ inventory($o$)
    \If{$s \bmod 4 = 0$} \Call{OwnReflexTick}{$o, s$} \EndIf
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Next action: execute the plan, refill it from the model when it runs dry}
\begin{algorithmic}[1]
\Procedure{NextAction}{$o, s$}
    \If{every milestone of $L$ is achieved} \Return noop \Comment{coast to the horizon} \EndIf
    \If{$Q = \langle\rangle$} \Call{ActTurn}{$o, s$} \EndIf
    \State $(\mathit{head}, i) \gets Q[0]$
    \If{$i \neq i_{\mathrm{cur}}$} \Comment{an entry boundary}
        \State $i_{\mathrm{cur}} \gets i$; $p_0 \gets p$; $\mathit{inv}_0 \gets \mathit{inv}$
        \State store.\textsc{open}(entries$[i]$, $i$, state($o$), frame $= o.\mathit{pov}$) \Comment{\file{episodes/ep\_$n$/}}
    \EndIf
    \If{$\mathit{head}$ is a closed-loop marker (servo, goto, dig, seq, click, \dots)}
        \State $a \gets \mathit{head}.\textsc{tick}(o)$; pop $\mathit{head}$ only when it reports done
    \Else
        \State $a \gets Q.\textsc{popleft}()$
    \EndIf
    \If{entry $i$ just ended} \Call{FinishEntry}{$i$} \EndIf
    \State \Return $a$
\EndProcedure
\Statex
\Procedure{FinishEntry}{$i$}
    \State store.\textsc{close}(events $= \mathit{pending}$, notes $= \mathit{notes}$) \Comment{\file{outcome.json}, one row of \file{index.jsonl}, tags}
    \Log{\texttt{[ACTION]} section: entry text, moved, descended, \texttt{[INV]} delta, \texttt{[EVENT]}s, \texttt{[NOTE]}s, frame path}
    \State $\mathit{outcomes} \gets \mathit{outcomes} \,\Vert\, \{\text{label}, \lVert p - p_0 \rVert, y_0 - y, \mathit{inv} - \mathit{inv}_0, \mathit{pending}, \text{first note}, \text{frame}\}$
    \State $\mathit{pending}, \mathit{notes} \gets \emptyset$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Act turn: one model call}
\begin{algorithmic}[1]
\Procedure{ActTurn}{$o, s$}
    \State $\mathit{stall} \gets 0$ \textbf{if} (milestones, inventory) changed since the last turn \textbf{else} $\mathit{stall} + 1$
    \State store.\textsc{flush}($s$) \Comment{an entry $Q$ dropped closes as \emph{flushed}}
    \State $f \gets$ save $o.\mathit{pov}$ as \file{attach/$s$.png}
    \State $\pi \gets$ \Call{TurnPrompt}{$o, s$}
    \State publish $W$ \Comment{windowed copy of \file{logs.txt}; frames linked}
    \For{$k = 1 \dots R$} \Comment{$R$ = analyzer retries}
        \State $r \gets T.\textsc{run}(\pi \text{ if } k = 1 \text{ else } \textsc{Nudge},\ \text{images} = [f])$
        \Statex \Comment{inside the call the model may grep \file{logs.txt}, run \file{tools/episodes.py}, \texttt{view\_image} any frame,}
        \Statex \Comment{read and write \file{mine/*}, write \file{mine/procedures/*.py} and \file{mine/reflexes/*.py}; it must write \file{actions.json}}
        \If{$r.\mathit{ok}$} \textbf{break} \EndIf
        \State sleep $\min(30, 2^k)$
    \EndFor
    \If{a harness-owned file changed} \Note{``not yours to write''} \EndIf; sweep foreign files
    \State $(\mathit{briefing}, \mathit{plan}) \gets$ split($r.\mathit{message}$)
    \State $\mathit{expect} \gets$ the \texttt{EXPECT:} line of $\mathit{briefing}$, if any
    \Log{\texttt{[PLAN]} $\mathit{plan}$} \Comment{repeated in every section until replaced}
    \State store.\textsc{plan}(turn, $\mathit{briefing}$, $\mathit{plan}$)
    \If{no \file{actions.json}}
        \Note{``no valid actions.json this turn; one no-op was executed''}; $Q \gets \langle \text{noop} \rangle$; \Return
    \EndIf
    \State $P \gets$ \Call{Parse}{\file{actions.json}} \Comment{$\le 5$ entries, $\le 400$ steps; entries are $\{a \in \mathcal{A}, \text{repeat}\}$ or $\{\text{own procedure}, \text{args}\}$}
    \Statex \Comment{an own procedure runs \texttt{python mine/procedures/$\langle$name$\rangle$.py} on the live state and expands to its atoms}
    \State \textbf{note} each parse note
    \State $Q \gets \langle (\sigma, i) : i, e \in \text{enumerate}(P.\mathit{entries}),\ \sigma \in e.\mathit{steps} \rangle$
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{The turn prompt: only what changed; the standing contract is \file{AGENTS.md}}
\begin{algorithmic}[1]
\Function{TurnPrompt}{$o, s$}
    \State $\mathit{head} \gets$ ``Step $s/N$, milestones $k/24$''; ``Act turns since anything verifiably changed: $\mathit{stall}$''
    \State $\mathit{head} \gets \mathit{head}$ $\Vert$ milestones block $\Vert$ \texttt{[STATE]} $p$, vitals, inventory, hotbar, held, nearby entities, GUI, cursor
    \State $\mathit{head} \gets \mathit{head}$ $\Vert$ \texttt{[MAP]} what this life's own positions and events prove
    \State $\mathit{head} \gets \mathit{head}$ $\Vert$ \texttt{[OWN]} \file{mine/procedures/*.py} with call counts; \file{mine/reflexes/*.py} with fire counts and which are off
    \State $\mathit{body} \gets$ \texttt{[EXPECTED]} $\mathit{expect}$ $\Vert$ \texttt{[OBSERVED]} one line per outcome in $\mathit{outcomes}$ (moved, descended, inv $\Delta$, events, note, frame)
    \State $\mathit{body} \gets \mathit{body}$ $\Vert$ ``Begin your briefing \texttt{OBSERVED vs EXPECTED: ...}; end it \texttt{EXPECT: ...}''
    \State $\mathit{body} \gets \mathit{body}$ $\Vert$ \texttt{[WORKING]} \file{mine/working.md} (first 1500 chars)
    \State $\mathit{body} \gets \mathit{body}$ $\Vert$ \texttt{[RECALL]} plan stalls, atom streaks, repeated notes, here-before (from \file{index.jsonl}; absent when empty)
    \State $\mathit{body} \gets \mathit{body}$ $\Vert$ ``Your memory is \file{./logs.txt}; \file{./tools/episodes.py} queries the episodes; \texttt{view\_image} opens a frame. The current view is attached. Write \file{./actions.json}.''
    \State \Return $\mathit{head} \,\Vert\, \mathit{body}$
\EndFunction
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{The model's own reflexes: the tick layer}
\begin{algorithmic}[1]
\Procedure{OwnReflexTick}{$o, s$} \Comment{every 4 ticks, outside a GUI}
    \State $\sigma \gets \{p, p_{\mathrm{prev}}, h, h_{\mathrm{prev}}, \text{food}, \text{food}_{\mathrm{prev}}, \text{lava\_frac}(\mathit{pov}), \mathit{dim}, \text{running entry}, |Q|, \mathit{inv}\}$
    \For{each live script $\rho$ \textbf{in} sorted(\file{mine/reflexes/*.py}), at most $M$, past its cooldown}
        \State $\mathit{res} \gets$ run \texttt{python} $\rho$ with $\sigma$ on stdin
        \If{$\rho$ failed $K$ times in a row} switch $\rho$ off until its file changes; \textbf{note} it; \textbf{continue} \EndIf
        \If{$\mathit{res} = \emptyset$} \textbf{continue} \EndIf
        \Note{``\texttt{[REFLEX} $\rho$\texttt{]} $\mathit{res}.\mathit{note}$''}
        \If{$\mathit{res}.\mathit{stop}$ or $\mathit{res}.\mathit{actions} \neq \emptyset$} $Q \gets \langle\rangle$ \EndIf
        \If{$\mathit{res}.\mathit{actions} \neq \emptyset$} $Q \gets$ \Call{Parse}{$\mathit{res}.\mathit{actions}$} \EndIf
        \State \Return \Comment{the first reflex that fires wins}
    \EndFor
\EndProcedure
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{What the model does with it (the contract, not harness code)}
\begin{algorithmic}[1]
\Statex \textbf{Turn 1 of a life with \file{previous/}: the workshop turn}
\State read \file{previous/life\_*/mine/\{digest,notes\}.md} and each life's death lines
\State query \file{tools/episodes.py --previous} \dots; view the frames that matter
\State write \file{mine/digest.md}: verified places (coordinates, life/step, episode id), what each life died of and where, which scripts and reflexes earned their keep, which assumptions broke
\State copy the scripts worth keeping into \file{mine/procedures/} and \file{mine/reflexes/}
\State end with a no-op plan \Comment{turns cost wall clock, not steps}
\Statex \textbf{Every later turn}
\State read \texttt{[OBSERVED]} against your own \texttt{EXPECT}; look at the cited frame when they differ
\State grep \file{logs.txt} or query the episodes when the present does not explain itself
\State rewrite \file{mine/working.md} when the plan changes ($< 40$ lines)
\State write \file{actions.json}: $\le 5$ entries of atoms or own procedures, with an \texttt{EXPECT} line
\State write a new procedure when the same atom sequence recurs; a new reflex when the same event keeps killing you; each firing into \file{mine/notes.md}
\Statex \textbf{Between lives}
\State nothing is summarised for you; the next life reads what you wrote under \file{mine/}
\end{algorithmic}
\end{algorithm}
